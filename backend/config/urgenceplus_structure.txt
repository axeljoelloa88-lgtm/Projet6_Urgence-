-- ============================================================
--  2. TYPES ENUM
-- ============================================================

CREATE TYPE role_enum                AS ENUM ('operateur', 'superviseur', 'admin');
CREATE TYPE vehicule_type_enum       AS ENUM ('ambulance', 'camion_pompier', 'voiture_patrouille');
CREATE TYPE vehicule_statut_enum     AS ENUM ('disponible', 'en_mission', 'hors_service');
CREATE TYPE urgence_type_enum        AS ENUM ('incendie', 'medical', 'accident', 'autre');
CREATE TYPE priorite_enum            AS ENUM ('faible', 'moyenne', 'haute', 'critique');
CREATE TYPE appel_statut_enum        AS ENUM ('recu', 'en_traitement', 'cloture');
CREATE TYPE intervention_statut_enum AS ENUM ('en_attente', 'en_route', 'sur_place', 'termine');


-- ============================================================
--  3. TABLES
-- ============================================================

-- TABLE : utilisateurs
CREATE TABLE utilisateurs (
    id            SERIAL       PRIMARY KEY,
    nom           VARCHAR(100) NOT NULL,
    email         VARCHAR(150) NOT NULL UNIQUE,
    mot_de_passe  VARCHAR(255) NOT NULL,    -- hashé avec bcrypt (10 rounds)
    role          role_enum    NOT NULL DEFAULT 'operateur',
    "createdAt"   TIMESTAMP    NOT NULL DEFAULT NOW(),
    "updatedAt"   TIMESTAMP    NOT NULL DEFAULT NOW()
);

-- TABLE : vehicules
CREATE TABLE vehicules (
    id               SERIAL               PRIMARY KEY,
    nom              VARCHAR(100)         NOT NULL,
    immatriculation  VARCHAR(20)          NOT NULL UNIQUE,
    type             vehicule_type_enum   NOT NULL,
    statut           vehicule_statut_enum NOT NULL DEFAULT 'disponible',
    latitude         FLOAT,
    longitude        FLOAT,
    "createdAt"      TIMESTAMP            NOT NULL DEFAULT NOW(),
    "updatedAt"      TIMESTAMP            NOT NULL DEFAULT NOW()
);

-- TABLE : appels
CREATE TABLE appels (
    id            SERIAL            PRIMARY KEY,
    appelant_nom  VARCHAR(100)      NOT NULL,
    appelant_tel  VARCHAR(20)       NOT NULL,
    description   TEXT,
    type_urgence  urgence_type_enum NOT NULL,
    priorite      priorite_enum     NOT NULL DEFAULT 'moyenne',
    latitude      FLOAT,
    longitude     FLOAT,
    adresse       VARCHAR(255),
    statut        appel_statut_enum NOT NULL DEFAULT 'recu',
    operateur_id  INT               REFERENCES utilisateurs(id) ON DELETE SET NULL,
    "createdAt"   TIMESTAMP         NOT NULL DEFAULT NOW(),
    "updatedAt"   TIMESTAMP         NOT NULL DEFAULT NOW()
);

-- TABLE : interventions
CREATE TABLE interventions (
    id           SERIAL                   PRIMARY KEY,
    statut       intervention_statut_enum NOT NULL DEFAULT 'en_attente',
    date_debut   TIMESTAMP,
    date_fin     TIMESTAMP,
    notes        TEXT,
    appel_id     INT  REFERENCES appels(id)       ON DELETE CASCADE,
    vehicule_id  INT  REFERENCES vehicules(id)    ON DELETE SET NULL,
    agent_id     INT  REFERENCES utilisateurs(id) ON DELETE SET NULL,
    "createdAt"  TIMESTAMP NOT NULL DEFAULT NOW(),
    "updatedAt"  TIMESTAMP NOT NULL DEFAULT NOW()
);

-- TABLE : rapports
CREATE TABLE rapports (
    id               SERIAL       PRIMARY KEY,
    titre            VARCHAR(200) NOT NULL,
    contenu          TEXT,
    intervention_id  INT  REFERENCES interventions(id) ON DELETE CASCADE,
    auteur_id        INT  REFERENCES utilisateurs(id)  ON DELETE SET NULL,
    "createdAt"      TIMESTAMP NOT NULL DEFAULT NOW(),
    "updatedAt"      TIMESTAMP NOT NULL DEFAULT NOW()
);


-- ============================================================
--  4. INDEX
-- ============================================================

CREATE INDEX idx_appels_statut        ON appels(statut);
CREATE INDEX idx_appels_priorite      ON appels(priorite);
CREATE INDEX idx_appels_operateur     ON appels(operateur_id);
CREATE INDEX idx_interventions_statut ON interventions(statut);
CREATE INDEX idx_interventions_appel  ON interventions(appel_id);
CREATE INDEX idx_vehicules_statut     ON vehicules(statut);
